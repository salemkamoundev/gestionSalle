  // --- AJOUT : LOGIQUE D'AFFICHAGE DES SERVICES (ONGLET INFO) ---
  // Récupération de tous les services pour l'affichage
  private allServicesSource = toSignal(this.serviceService.getAll(), { initialValue: [] });

  // Calcul des services sélectionnés
  selectedServicesDisplay = computed(() => {
    const selectedIds = this.form.get('services')?.value || [];
    const list = this.allServicesSource();
    if (!list || list.length === 0) return [];
    
    // Filtre les services dont l'ID est coché
    return list.filter(s => selectedIds.includes(s.id));
  });

  // Calcul du total pour l'affichage
  getServicesTotalDisplay(): number {
    return this.selectedServicesDisplay().reduce((acc, s) => acc + (Number(s.price) || 0), 0);
  }
